STANDARD INTEGRATION PATTERN:
```python
from src.memory.long_term import VectorMemory
from src.memory.episodic import EpisodicMemory

class YourAgent:
    def __init__(self, agent_id: str):
        # Each agent gets own collection
        self.memory = VectorMemory(collection_name=f"agent_{agent_id}")
        self.episodic = EpisodicMemory()
    
    def handle(self, query: str, context: dict = None):
        # 1. Retrieve memories
        memories = self.memory.search(query, limit=5)
        recent = self.episodic.get_recent(agent=self.agent_id, limit=10)
        
        # 2. Process with memories
        response = self.process(query, memories, recent, context)
        
        # 3. Store new memories
        self.memory.add(
            text=f"Q: {query}\nA: {response}",
            metadata={"agent": self.agent_id, "timestamp": datetime.now().isoformat()}
        )
        self.episodic.log(
            agent=self.agent_id,
            event_type="interaction",
            data={"query": query, "response": response}
        )
        
        return response
```

- VectorMemory.add(text, metadata) → None
- EpisodicMemory.log(agent, event_type, data) → None
- EpisodicMemory.get_recent(agent, limit) → List[dict]

```python
class InitializationError(AgentError):

class MemoryError(AgentError):

```

STANDARD PATTERN:
```python
import logging
from typing import Optional, Dict, Any

logger = logging.getLogger(__name__)

class YourAgent:
    def handle(self, query: str, context: Dict = None) -> Optional[Dict[str, Any]]:
        """Handle request with comprehensive error handling."""
        
        # Validate input
        if not query or not isinstance(query, str):
            logger.warning(f"{self.name}: Invalid query")
            return {
                'success': False,
                'error': 'Invalid query format',
                'error_type': 'validation_error'
            }
        
        try:
            # Main processing
            result = self.process(query, context)
            return {
                'success': True,
                'result': result,
                'agent': self.name
            }
            
        except ProcessingError as e:
            logger.error(f"{self.name} processing error: {e}")
            return {
                'success': False,
                'error': 'Processing failed',
                'error_type': 'processing_error',
                'fallback': self.get_fallback_response(query)
            }
            
        except MemoryError as e:
            logger.error(f"{self.name} memory error: {e}")
            # Continue without memory if necessary
            return self.process_without_memory(query, context)
            
        except Exception as e:
            logger.exception(f"{self.name} unexpected error: {e}")
            return {
                'success': False,
                'error': 'Internal error occurred',
                'error_type': 'unexpected_error',
                'fallback': self.get_safe_fallback(query)
            }
```

FALLBACK STRATEGIES:
```python
def get_fallback_response(self, query: str) -> str:
    """Multi-level fallback strategy."""
    
    # Level 1: Try simplified processing
    try:
        return self.simple_process(query)
    except Exception:
        pass
    
    # Level 2: Rule-based response
    try:
        return self.rule_based_response(query)
    except Exception:
        pass
    
    # Level 3: Template response
    return f"I encountered an issue processing your request. Please try rephrasing."
```

```python
# tests/run_all_tests.py
import subprocess
import sys

def run_tests():
    """Run all test suites."""
    
    test_suites = [
        ("Unit Tests", "pytest tests/agents/ -v"),
        ("Integration Tests", "pytest tests/integration/ -v"),
        ("Performance Tests", "pytest tests/performance/ -v"),
    ]
    
    all_passed = True
    
    for name, command in test_suites:
        print(f"\n{'='*60}")
        print(f"Running: {name}")
        print('='*60)
        
        result = subprocess.run(command.split(), capture_output=True)
        
        if result.returncode != 0:
            print(f"❌ {name} FAILED")
            all_passed = False
        else:
            print(f"✅ {name} PASSED")
    
    return 0 if all_passed else 1

if __name__ == "__main__":
    sys.exit(run_tests())
```

```python
class SharedContext:
    def __init__(self):
        self.state = {}
        self.lock = threading.Lock()
    
    def update(self, key: str, value: Any):
        with self.lock:
            self.state[key] = value
    
    def get(self, key: str) -> Any:
        with self.lock:
            return self.state.get(key)
```
